DP Direction Notes

1. Climbing Stairs

A. Recursion (0 -> n)

    int solve(int i,int n){
        if(i==n) return 1;
        if(i>n) return 0; 
        return solve(i+1,n)+solve(i+2,n);
    }

Tabulation

    vector<int> dp(n+2,0);
    dp[n]=1;
    for(int i=n-1;i>=0;i--)
        dp[i]=dp[i+1]+dp[i+2];

------------------------------------------------------------------------

B. Recursion (n -> 0)

    int solve(int i){
        if(i==0) return 1;
        if(i<0) return 0;
        return solve(i-1)+solve(i-2);
    }

Tabulation

    vector<int> dp(n+1,0);
    dp[0]=1;
    for(int i=1;i<=n;i++){
        int one=dp[i-1];
        int two=(i>=2)?dp[i-2]:0;
        dp[i]=one+two;
    }

2. Min Cost Climbing Stairs

A. Recursion (0 -> n)

    int solve(int i,vector<int>& cost){
        if(i>=cost.size()) return 0;
        int one=cost[i]+solve(i+1,cost);
        int two=cost[i]+solve(i+2,cost);
        return min(one,two);
    }

Tabulation

    vector<int> dp(n+2,0);
    for(int i=n-1;i>=0;i--)
        dp[i]=cost[i]+min(dp[i+1],dp[i+2]);

    return min(dp[0],dp[1]);

------------------------------------------------------------------------

B. Recursion (n -> 0)

    int solve(int i,vector<int>& cost){
        if(i==0) return cost[0];
        if(i==1) return cost[1];
        return cost[i]+min(solve(i-1,cost),solve(i-2,cost));
    }

Tabulation

    vector<int> dp(n);
    dp[0]=cost[0];
    dp[1]=cost[1];

    for(int i=2;i<n;i++)
        dp[i]=cost[i]+min(dp[i-1],dp[i-2]);

    return min(dp[n-1],dp[n-2]);

3. Unique Paths

Recursion (0 -> destination)

    int solve(int i,int j,int m,int n){
        if(i==m-1 && j==n-1) return 1;
        if(i>=m || j>=n) return 0;
        return solve(i,j+1,m,n)+solve(i+1,j,m,n);
    }

Tabulation

    vector<vector<int>> dp(m,vector<int>(n,0));
    dp[m-1][n-1]=1;

    for(int i=m-1;i>=0;i--){
        for(int j=n-1;j>=0;j--){
            if(i==m-1 && j==n-1) continue;

            int right=(j+1<n)?dp[i][j+1]:0;
            int down=(i+1<m)?dp[i+1][j]:0;

            dp[i][j]=right+down;
        }
    }

Golden Rule

-   Depends on i+1/i+2 -> Tabulation: last to first.
-   Depends on i-1/i-2 -> Tabulation: first to last.
